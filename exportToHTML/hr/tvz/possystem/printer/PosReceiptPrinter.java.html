    /**
     * Ispisuje QR kod na POS termalnom pisaču koristeći ESC/POS naredbe.
     * Zpisuje sadržaj QR koda u pisač preko predanog izlaznog toka.
     *
     * @param data tekstualni sadržaj koji se kodira u QR kod
     * @param os   izlazni tok povezan s POS pisačem
     * @throws IOException ako dođe do pogreške pri pisanju u izlazni tok
     */
    private static void printQrWithEscPos(String data, OutputStream os) throws IOException {
        os.write(new byte[]{0x1D, 0x28, 0x6B, 0x04, 0x00, 0x31, 0x41, 0x32, 0x00});
        os.write(new byte[]{0x1D, 0x28, 0x6B, 0x03, 0x00, 0x31, 0x43, 0x04});
        os.write(new byte[]{0x1D, 0x28, 0x6B, 0x03, 0x00, 0x31, 0x45, 0x31});

        byte[] payload = data.getBytes(java.nio.charset.StandardCharsets.ISO_8859_1);
        int len = payload.length + 3;
        byte pL = (byte) (len & 0xFF);
        byte pH = (byte) ((len >> 8) & 0xFF);
        os.write(new byte[]{0x1D, 0x28, 0x6B, pL, pH, 0x31, 0x50, 0x30});
        os.write(payload);
        os.write(new byte[]{0x1D, 0x28, 0x6B, 0x03, 0x00, 0x31, 0x51, 0x30});

        os.flush();
    }
